int def SwitchDemo(int param choice)
{
	print "--- Демонстрация switch/case ---";
	switch(choice)
	{
		case(1):
		{
			print "Выбран Кейс 1: Проверка целочисленных литералов";
			
			int let binaryVal;
			binaryVal = b1101;
			int let octalVal;
			octalVal = h33;
			
			print "Двоичный b1101 (13): ";
			print binaryVal;
			print "Восьмеричный h33 (27): ";
			print octalVal;
		}
		case(2):
		{
			print "Выбран Кейс 2: Проверка функций времени и случайных чисел";
			
			str let rTime; rTime = getLocalTimeAndDate();
			print "Текущее время: ";
			print rTime;

			int let rResult; rResult = random(1, 10);
			print "Случайное число от 1 до 10: ";
			print rResult;

			break;
		}
		default:
		{
			print "Дефолтный Кейс: Неверный выбор";
			break;
		}
	}
	return choice;
}
