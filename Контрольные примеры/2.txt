int def firstTrySwitch(int param testIntVariable)
{
	switch(testIntVariable)
	{
		case(1):
		{
			print "Кейс номер 1";
			break;
		}
		case(2):
		{
			print "Кейс номер 2";
			break;
		}
	}
	return testIntVariable;
}


int def variableCaseSwitch(int param value, int param threshold)
{
	int let result = 0;
	switch(value)
	{
		case(threshold):
		{
			print "Сработал кейс с переменной threshold";
			result = 1;
			break;
		}
		default:
		{
			print "Дефолт";
			result = 0;
			break;
		}
	}
	return result;
}

bool def boolVariableCase(bool param value)
{
	switch(value)
	{
		case(1):
		{
			print "Кейс true";
			break;
		}
		case(0):
		{
			print "Кейс false";
			break;
		}
		case(2):
		{
			print "Кейс 2, ошибка";
			break;
		}
	}
	return value;
}

main
{
	# Тестирование switch-case 
	str let testStringVariable;
	int let testIntVariable;
	int let threshold;
	bool let testBoolVariable;

	print "--- Тест 1: Switch без default ---";
	print "ТЕСТ 1,1 (пробуем кейс 1, результат должен быть Кейс номер 1)";
	testIntVariable = 1;
	testIntVariable = firstTrySwitch(testIntVariable);

	print "ТЕСТ 1,2 (пробуем кейс 2, результат должен быть Кейс номер 2)";
	testIntVariable = 2;
	testIntVariable = firstTrySwitch(testIntVariable);

	print "ТЕСТ 1,3 (пробуем кейс 3, в результате ничего не должно вывестись, такого кейса нет)";
	testIntVariable = 3;
	testIntVariable = firstTrySwitch(testIntVariable);

	print "--- Тест 2: Switch с default, case переменной---";
	print "ТЕСТ 2,1 (пробуем кейс по переменной, в результате должно вывестись, Сработал кейс с переменной threshold)";
	threshold = 7;
	testIntVariable = 7;
	testIntVariable = variableCaseSwitch(testIntVariable, threshold);

	print "ТЕСТ 2,1 (пробуем вызвать дефолт, в результате должно вывестись, Дефолт)";
	threshold = 7;
	testIntVariable = 8;
	testIntVariable = variableCaseSwitch(testIntVariable, threshold);

	print "--- Тест 3: Switch с переменной bool---";
	
	print "ТЕСТ 3,1 (пробуем найти кейс по переменной true)";
	testBoolVariable = true;
	testBoolVariable = boolVariableCase(testBoolVariable);
	
	print "ТЕСТ 3,2 (пробуем найти кейс по переменной false)";
	testBoolVariable = false;
	testBoolVariable = boolVariableCase(testBoolVariable);
}
